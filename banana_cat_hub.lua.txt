-- Drag logic
local dragging   = false
local dragStart  = nil
local startPos   = nil
local hasDragged = false

LogoButton.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
    or input.UserInputType == Enum.UserInputType.Touch then
        dragging   = true
        hasDragged = false
        dragStart  = input.Position
        startPos   = LogoButton.Position
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if not dragging then return end
    if input.UserInputType == Enum.UserInputType.MouseMovement
    or input.UserInputType == Enum.UserInputType.Touch then
        local delta = input.Position - dragStart
        if delta.Magnitude > 6 then
            hasDragged = true
        end
        LogoButton.Position = UDim2.new(
            startPos.X.Scale,
            startPos.X.Offset + delta.X,
            startPos.Y.Scale,
            startPos.Y.Offset + delta.Y
        )
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
    or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)

LogoButton.MouseButton1Click:Connect(function()
    if not hasDragged then
        Window.Root.Visible = not Window.Root.Visible
    end
end)

LogoButton.MouseEnter:Connect(function()
    LogoLabel.Visible = true
    LogoButton.BackgroundColor3 = Color3.fromRGB(40, 35, 10)
end)

LogoButton.MouseLeave:Connect(function()
    LogoLabel.Visible = false
    LogoButton.BackgroundColor3 = Color3.fromRGB(25, 25, 25)
end)

UserInputService.InputBegan:Connect(function(Input, Processed)
    if Processed then return end
    if Input.KeyCode == Enum.KeyCode.LeftControl then
        Window.Root.Visible = not Window.Root.Visible
    end
end)

-- ESP
local ESPObjects = {}

local function RemoveESP(Player)
    if ESPObjects[Player] then
        ESPObjects[Player]:Destroy()
        ESPObjects[Player] = nil
    end
end

local function CreateESP(Player)
    if Player == LocalPlayer then return end
    if ESPObjects[Player] then return end
    local Character = Player.Character
    if not Character then return end

    local Highlight = Instance.new("Highlight")
    Highlight.Name                = "BananaCatESP"
    Highlight.FillTransparency    = 0.8
    Highlight.OutlineTransparency = 0
    Highlight.DepthMode           = Enum.HighlightDepthMode.AlwaysOnTop
    Highlight.Parent              = Character
    ESPObjects[Player]            = Highlight
end

local function RefreshESP()
    for _, Player in ipairs(Players:GetPlayers()) do
        if Player ~= LocalPlayer then
            if Options.PlayerESP.Value then CreateESP(Player)
            else RemoveESP(Player) end
        end
    end
end

Tabs.ESP:AddToggle("PlayerESP",  { Title = "Player ESP",  Default = false })
Tabs.ESP:AddToggle("FruitESP",   { Title = "Fruit ESP",   Default = false })

Tabs.ESP:AddToggle("IslandESP",  { Title = "Island ESP",  Default = false })

Options.PlayerESP:OnChanged(function() RefreshESP() end)

Players.PlayerAdded:Connect(function(Player)
    Player.CharacterAdded:Connect(function()
        task.wait(0.5)
        if Options.PlayerESP.Value then CreateESP(Player) end
    end)
end)

Players.PlayerRemoving:Connect(function(Player) RemoveESP(Player) end)

Tabs.Webhook:AddButton({
    Title    = "Send Test Webhook",
    Callback = function()
        local URL     = Options.WebhookURL.Value
        local Message = Options.WebhookMessage.Value

        if URL == "" then
            Notify("Webhook URL is empty.")
            return
        end

        if not request then
            Notify("HTTP request function is unavailable.")
            return
        end

        local Body = HttpService:JSONEncode({ content = Message })

        local Success, Err = pcall(function()
            request({
                Url    = URL,
                Method = "POST",
                Headers = { ["Content-Type"] = "application/json" },
                Body   = Body
            })
        end)

        if Success then
            Notify("Webhook sent!")
        else
            Notify("Failed: " .. tostring(Err))
        end
    end
})
